# Kubernetes 指令集

* ctr 是 containerd 指令，类似 docker 指令，最小操作单元是容器
* crictl 是Pod指令，是CRI兼容的容器指令。最小操作单元是pod

|                   |                   |      |
|-------------------|-------------------|------|
| crictl ps         | docker ps         |      |
| crictl images     | docker images     |      |
| crictl logs       | docker logs       |      |
| crictl inspect    | docker inspect    |      |
| crictl stats      | docker stats      |      |
| crictl start/stop | docker start/stop |      |
| crictl create     | docker create     | 创建容器 |
| crictl rm/rmi     | docker rm/rmi     |      |
| crictl pull       | docker pull       |      |
| crictl exec       | docker exec       |      |

|                  |             |                |
|------------------|-------------|----------------|
| ctr run          | docker run  | 运行一个容器，而不是 pod |
| ctr image tag    | docker tag  |                |
| ctr image import | docker load | 导入镜像           |
| ctr image export | docker save | 导出镜像           |
| ctr image push   | docker push |                |
|                  |             |                |   

```shell
ctr -v      # 查看containerd版本
crictl -v   # 查看kubernetes版本
crictl pods # 列出所有pods

```

## 关闭Swap

## 加载内核和网络模块

```shell
cat <<EOF | sudo tee /etc/modules-load.d/k3s-cilium.conf
overlay
br_netfilter
EOF

# 加载OverlayFS内核模块，即 overlay.ko
sudo modprobe overlay
# 加载 br_netfilter 内核模块，用于桥接网络流量的过滤和转发，需配合 sysctl 设置如 net.bridge.bridge-nf-call-iptables=1 以启用桥接流量过滤
sudo modprobe br_netfilter

# 验证模块加载
lsmod | grep -E "overlay|br_netfilter"

# 检测是否支持IPv6请求，如果不支持，则禁止IPv6；反之不用禁止
ping6 -c 4 registry-1.docker.io
ping6 -c 4 google.com

# 不支持IPv6情况
sudo tee /etc/sysctl.d/99-disable-ipv6.conf << EOF
# Disable IPv6
net.ipv6.conf.all.disable_ipv6 = 1
net.ipv6.conf.default.disable_ipv6 = 1
net.ipv6.conf.lo.disable_ipv6 = 1
EOF


# enable IPv4 forward
sudo tee /etc/sysctl.d/99-kubernetes-cilium.conf <<EOF
net.ipv4.ip_forward = 1
net.ipv4.conf.all.forwarding = 1
net.bridge.bridge-nf-call-iptables = 1
EOF

sudo sysctl --system
# check net.ipv4.ip_forward=1
sudo sysctl net.ipv4.ip_forward


# 查看本机器 product_uuid
sudo cat /sys/class/dmi/id/product_uuid
```

## 安装 kubelet kubeadm kubectl

```shell
sudo mkdir -p /etc/apt/keyrings/
K8S_VERSION=1.34
curl -fsSL "https://pkgs.k8s.io/core:/stable:/v${K8S_VERSION}/deb/Release.key" | sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
echo "deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v${K8S_VERSION}/deb/ /" | sudo tee /etc/apt/sources.list.d/kubernetes.list
sudo apt update
sudo apt install -y kubelet kubeadm kubectl
# 锁定版本
sudo apt-mark hold kubelet kubeadm kubectl
```


## 控制面板 kubeadm init
在控制面板(Master Node)上，进行`kubeadm init`初始化。
```shell
# 如果/etc/crictl.yaml不存在，则需要创建
cat <<EOF | sudo tee /etc/crictl.yaml
runtime-endpoint: unix:///var/run/containerd/containerd.sock
image-endpoint: unix:///var/run/containerd/containerd.sock
timeout: 10
debug: false
EOF


# 关闭VPN
unset http_proxy
unset https_proxy
# 预先使用这个拉取核心镜像，减少 kubeadm init 实时拉取时间
sudo kubeadm config images pull --cri-socket=unix:///var/run/containerd/containerd.sock --image-repository=registry.aliyuncs.com/google_containers
# 检查拉取的镜像
sudo crictl images
# 查看所有 Kubernetes 相关的容器
sudo crictl --runtime-endpoint unix:///var/run/containerd/containerd.sock ps -a

sudo apt install -y net-tools
# 检查端口占用
sudo netstat -tulnp | grep 10250

# 使用 Cilium cidr
# --upload-certs 用于自动上传和控制平面证书，以便其他控制平面节点可以安全地加入集群。
# 这个很耗时，需要耐心等待
# 重要！ 命令成功运行后，会输出 `kubeadm join ....` 命令。请务必复制并保存这个命令，后续需要用它来将工作节点加入集群。
#sudo kubeadm init --config=kubeadm.yaml
sudo kubeadm init \
  --pod-network-cidr=10.0.0.0/16  \
  --service-cidr=10.96.0.0/12     \
  --image-repository=registry.aliyuncs.com/google_containers \
  --upload-certs \
  --v=5

# 安装输入的操作配置kubectl权限
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config
echo "export KUBECONFIG=$HOME/.kube/config" >> ~/.bashrc
source ~/.bashrc
```


## 安装Cilium CNI
这里安装 Cilium CNI网络插件，使用的是 cilium cli，就不用在使用  kubectl apply -f xxx.yml 安装了

```shell
# 安装 cilium cli
cd ~/Downloads
CILIUM_CLI_VERSION=$(curl -s https://raw.githubusercontent.com/cilium/cilium-cli/main/stable.txt)
echo $CILIUM_CLI_VERSION
curl -L --fail --remote-name-all https://github.com/cilium/cilium-cli/releases/download/${CILIUM_CLI_VERSION}/cilium-linux-amd64.tar.gz
sudo tar xzvfC cilium-linux-amd64.tar.gz /usr/local/bin
rm cilium-linux-amd64.tar.gz

# 安装 cilium
cilium install

# 验证 cilium 
cilium status --wait

# 查看 cilium 日志
kubectl logs -n kube-system -l k3s-app=cilium

# 检查 Cilium pods
kubectl get pods -n kube-system -l k3s-app=cilium

sudo chmod +x /opt/cni/bin/*
# dashboard 需要
ln -s /opt/cni/bin/cilium-cni /usr/lib/cni/cilium-cni

# 检查cni是否安装成功
ls -la /opt/cni/bin/loopback
```

## 安装 bash complete
```shell 
# install bash completion
sudo apt install -y bash-completion
echo 'source /usr/share/bash-completion/bash_completion' >>~/.bashrc
echo 'source <(kubectl completion bash)' >>~/.bashrc
# set alias
echo 'alias k=kubectl' >>~/.bashrc
echo 'complete -o default -F __start_kubectl k' >>~/.bashrc
```
