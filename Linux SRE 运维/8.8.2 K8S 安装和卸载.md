# 卸载Kubernetes

## 重置kubelet
如果之前执行过 kubeadm init ...  然后出错了，则需要清理后，才能执行
```shell
kubectl get nodes
systemctl status kubelet
sudo journalctl -u kubelet -f

sudo systemctl stop kubelet
sudo systemctl stop containerd
sudo kubectl drain k3s-master --delete-emptydir-data --force --ignore-daemonsets
sudo kubeadm reset -f
sudo iptables -F 
sudo iptables -t nat -F
sudo iptables -t mangle -F
sudo iptables -X
sudo kubectl delete node k3s-master
sudo systemctl daemon-reload
sudo systemctl start containerd
sudo systemctl status containerd
sudo shutdown -r now

# kubeadm init 报错，就查看 kubelet 日志
sudo journalctl -u kubelet -n 50 --no-pager
```


## 卸载Kubernetes

```shell
sudo systemctl stop kubelet
sudo systemctl stop containerd
sudo kubectl drain k3s-master --delete-emptydir-data --force --ignore-daemonsets
# 重置 kubeadm，清理节点
sudo kubeadm reset -f

# 清理网络规则和配置
sudo iptables -F 
sudo iptables -t nat -F
sudo iptables -t mangle -F
sudo iptables -X
sudo rm -rf /etc/cni/net.d

sudo kubectl delete node k3s-master



# 移除所有 Kubernetes 相关软件包
sudo apt-mark unhold kubelet kubeadm kubectl
sudo apt purge -y kubeadm kubectl kubelet kubernetes-cni cri-tools
sudo sudo apt autoremove -y
sudo apt clean
sudo apt autoclean -y

# 删除所有文件夹
sudo rm -rf /opt/cni
sudo rm -rf /etc/cni
sudo rm -rf ~/.kube
sudo rm -rf /etc/kubernetes
sudo rm -rf /var/lib/kubelet
sudo rm -rf /var/lib/etcd
sudo rm -rf /var/lib/cni
sudo rm -f /etc/sysctl.d/kubernetes.conf

# 卸载 containerd
sudo apt purge -y containerd 
sudo apt purge -y containerd.io
sudo rm -rf /var/lib/containerd
sudo rm -rf /etc/containerd

# 卸载 docker
sudo apt purge -y docker-ce docker-ce-cli docker-buildx-plugin docker-compose-plugin

# 移除 Kubernetes 源列表和API密钥
sudo rm -f /etc/apt/sources.list.d/kubernetes.list
sudo rm -f /usr/share/keyrings/kubernetes-archive-keyring.gpg
sudo apt update


# 清理残留进程
sudo pkill -f kubelet
sudo pkill -f kube-apiserver
sudo pkill -f kube-controller-manager
sudo pkill -f kube-scheduler
sudo pkill -f etcd

sudo reboot
```


# Kubernetes Master 节点安装其他



## 安装 Dashboard

https://kubernetes.io/zh-cn/docs/tasks/access-application-cluster/web-ui-dashboard/

```shell
# 卸载之前安装的 dashboard
kubectl delete -f https://raw.githubusercontent.com/kubernetes/dashboard/v2.7.0/aio/deploy/recommended.yaml


kubectl get namespaces
# 若 kubernetes-dashboard 状态是 Terminating，则需要强制删除卡住的namespace
cd ~/Downloads
kubectl get namespace kubernetes-dashboard -o json > dashboard-namespace.json
# 删除 spec.finalizers，即仅保留 spec: {}
sed -i '/"finalizers": \[/,/\]/d' dashboard-namespace.json
kubectl replace --raw "/api/v1/namespaces/kubernetes-dashboard/finalize" -f dashboard-namespace.json
kubectl get namespaces


# 安装 helm
curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash
helm version
# 添加 kubernetes-dashboard 仓库
helm repo add kubernetes-dashboard https://kubernetes.github.io/dashboard/
# 使用 kubernetes-dashboard Chart 部署名为 `kubernetes-dashboard` 的 Helm Release
helm upgrade --install kubernetes-dashboard kubernetes-dashboard/kubernetes-dashboard --create-namespace --namespace kubernetes-dashboard
# 此时kubernetes-dashboard 状态是 Active
kubectl get namespaces


helm list -n kubernetes-dashboard
kubectl get svc -n kubernetes-dashboard

# 创建管理员
kubectl apply -f - <<EOF
apiVersion: v1
kind: ServiceAccount
metadata:
  name: admin-user
  namespace: kubernetes-dashboard
EOF

# 创建集群管理员绑定
kubectl apply -f - <<EOF
apiVersion: rbac.authorization.k8s.io/v1
kind: ClusterRoleBinding
metadata:
  name: admin-user
roleRef:
  apiGroup: rbac.authorization.k8s.io
  kind: ClusterRole
  name: cluster-admin
subjects:
- kind: ServiceAccount
  name: admin-user
  namespace: kubernetes-dashboard
EOF


# 如果有报错，就修复。如果事 docker.io image 无法下载，就尝试手动下载
kubectl describe pods -n kubernetes-dashboard

# 获取访问令牌
kubectl -n kubernetes-dashboard create token admin-user


# 启动端口转发， https://192.168.0.250:8443 
kubectl -n kubernetes-dashboard port-forward svc/kubernetes-dashboard-kong-proxy 8443:443


```

## 安装 metrics-server 资源监控

```shell
kubectl apply -f https://github.com/kubernetes-sigs/metrics-server/releases/latest/download/components.yaml
kubectl patch service -n kube-system metrics-server --type='json' -p='[{"op": "replace", "path": "/spec/ports/0/name", "value": "https"}]'

# 检查 API 服务状态
kubectl describe apiservice v1beta1.metrics.k3s.io
kubectl top nodes

# 查看日志
kubectl logs -n kube-system -l k3s-app=metrics-server
```

## 常见排查错误
```shell
# 检查节点详情
kubectl describe node k3s-worker01

# 检查 kubelet 日志
sudo journalctl -u kubelet -f --no-pager

# 检查容器运行时日志
sudo journalctl -u containerd -f --no-pager
```

```shell

sudo systemctl enable --now kubelet


# kubelet 现在每隔几秒就会重启，因为它陷入了一个等待 kubeadm 指令的死循环。
# 检查 6443 端口是否开放
nc 127.0.0.1 6443 -zv -w 2

# 如果无法启动，则查看
sudo journalctl -u kubelet -f

# 不存在 /etc/kubenetes/pki 则执行
sudo kubeadm init phase certs all

# 如果不存在 config.yaml，就使用下面命令重新生成
sudo kubeadm init phase kubeconfig all
sudo kubeadm init phase kubelet-start --config=/etc/kubernetes/kubeadm-config.yaml
sudo mv /var/lib/kubelet/* /data/lib/kubelet

 
```
